import io
from django.core.files.base import ContentFile
from django.core.files.storage import default_storage



# save the plot to a file
buf = io.BytesIO()
plt.savefig(buf, format='png')
plt.close()
buf.seek(0)

plot_img_path = f'plots/{ticker}_plot.png'  # optional folder in S3
file_content = ContentFile(buf.read())
default_storage.save(plot_img_path, file_content)

# Get the public URL
plot_img_url = default_storage.url(plot_img_path)

return Response({
    'plot_img':plot_img_url,
    })
